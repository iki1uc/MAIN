const SECTORS = 12; const STEP_DEG = 360 / SECTORS; // 30° function sectorCenter(index, offsetDeg = 0) { return (offsetDeg + index * STEP_DEG + STEP_DEG / 2) % 360; } function normalizeDegrees(deg) { return ((deg % 360) + 360) % 360; } function angleDifference(a, b) { // kleinster gerichteter Abstand: Ergebnis in [-180°, 180°) return ((normalizeDegrees(a - b) + 540) % 360) - 180; } function sectorIndex(angleDeg, offsetDeg = 0) { return Math.floor( normalizeDegrees(angleDeg - offsetDeg + STEP_DEG / 2) / STEP_DEG ) % SECTORS; }
